\begin{document}
\begin{tikzpicture}[
  font=\small,
  every node/.style={text=ink},
  stage/.style={draw, rounded corners=2.5pt, align=left, text width=7.6cm,
                inner xsep=8pt, inner ysep=6pt, line width=0.7pt},
  preblock/.style={stage, draw=estc!75!black, fill=estc!7},
  runblock/.style={stage, draw=spat!75!black, fill=spat!8},
  stopblock/.style={stage, draw=freq!75!black, fill=freq!7},
  audblock/.style={stage, draw=advc, fill=black!6},
  verblock/.style={stage, draw=okgreen!80!black, fill=okgreen!8},
  ghost/.style={stage, draw=muted, dashed, fill=black!2, text=ink2},
  flow/.style={-{Latex[length=2.6mm]}, line width=0.8pt, draw=ink2},
  ghostflow/.style={-{Latex[length=2.4mm]}, line width=0.7pt, draw=muted, dashed},
  gpu/.style={font=\footnotesize, text=muted, anchor=west},
]

% ---------------------------------------------------------------- stages ----
\node[preblock] (prereg) at (0,0) {%
  {\head Pre-registration (before any result)}\\[1pt]
  {\body STOP-gate + paired-$\Delta$ win-rule frozen under a git tag;
   each later re-test's decision rule journaled before its run.}};

\node[runblock, below=5.5mm of prereg] (campaign) {%
  {\head Controlled campaign}\\[1pt]
  {\body full CAFRL (11.97\,M) vs capacity-matched EfficientNet-B0
   ``b7'' (4.0\,M) + Xception / F3-Net / FreqNet, all on identical
   multi-CRF data.\\
   \textit{Confounds caught \& ruled out:} augmentation policy;
   backbone capacity.}};

\node[stopblock, below=5.5mm of campaign] (stop) {%
  {\head STOP-gate fires --- the thesis is falsified}\\[1pt]
  {\body b7 beats full CAFRL at every level
   (c40: 0.959 vs 0.922; $\Delta$ +0.037, CI [+0.021,\,+0.054]).}};

\node[ghost, below right=4.5mm and -49mm of stop, text width=5.7cm] (ablations) {%
  {\head\color{ink2} Ablation campaign A1--A6}\\[1pt]
  {\body GRL-vs-HSIC bake-off, conditioning, fusion, curriculum arms
   --- \textbf{never run} per the frozen rule.}};

\node[audblock, anchor=north] (audit) at ($(stop.south -| stop.center)+(0,-3.6cm)$) {%
  {\head Adversarial self-audit of the negative}\\[1pt]
  {\body measurement reproduces bit-exactly; causal reading unsound.
   Four defects biased \emph{against} frequency: D1 auxiliary-loss
   dominance (40--120$\times$) $\cdot$ D2 $\gamma$ gradient deadlock $\cdot$
   D3 20$\times$ optimization mismatch $\cdot$ D4 bottlenecked trunk.}};

\node[runblock, below=5.5mm of audit] (retests) {%
  {\head Pre-registered fair re-tests (repair D1--D4)}\\[1pt]
  {\body T: $\approx$88\% of the gap was recipe (significant)\\
   F1 (fair recipe, zero invariance losses): 0.9586${}\approx{}$b7~0.9590\\
   F2 (full trunk + $\gamma$-residual): +0.0056, CI includes 0}};

\node[runblock, below=5.5mm of retests] (seeds) {%
  {\head Seed confirmation (3 seeds vs 3 seeds)}\\[1pt]
  {\body pooled $\Delta$ at c40: F1 $-$0.0010 / F2 +0.0022,
   both CIs include 0 --- neutral, confirmed.}};

\node[verblock, below=5.5mm of seeds] (verdict) {%
  {\head Verdict}\\[1pt]
  {\body deficit = recipe artifact (+3.96\,pt recovered);
   frequency informative alone (AUC 0.91--0.98) yet redundant in fusion;
   robustness is bought by data diversity (real CRF $\gg$ JPEG, +7.3\,pt).}};

% ----------------------------------------------------------------- edges ----
\draw[flow] (prereg.south) -- (campaign.north);
\draw[flow] (campaign.south) -- (stop.north);
\draw[ghostflow] (stop.south -| ablations.north) -- (ablations.north);
\draw[flow] ([xshift=5mm]stop.south west) -- ([xshift=5mm]audit.north west);
\draw[flow] (audit.south) -- (retests.north);
\draw[flow] (retests.south) -- (seeds.north);
\draw[flow] (seeds.south) -- (verdict.north);

% --------------------------------------------------------- GPU-h economics --
\node[gpu, text=freq!70!black, align=left] at ($(ablations.east)+(0.25,0)$)
  {$\approx$270 GPU-h\\[-2pt]\textbf{saved}};

  {$\approx$35 GPU-h};
\node[gpu, align=left] at ($(seeds.east)+(0.25,0)$)
  {$\approx$31 GPU-h};
