\begin{document}
\begin{tikzpicture}[
  font=\small,
  every node/.style={text=ink},
  stage/.style={draw, rounded corners=2.5pt, align=left, text width=39mm,
                inner xsep=6pt, inner ysep=5pt, line width=0.7pt,
                minimum height=30mm, anchor=north},
  preblock/.style={stage, draw=estc!75!black, fill=estc!7},
  runblock/.style={stage, draw=spat!75!black, fill=spat!8},
  stopblock/.style={stage, draw=freq!75!black, fill=freq!7},
  audblock/.style={stage, draw=advc, fill=black!6},
  verblock/.style={stage, draw=okgreen!80!black, fill=okgreen!8},
  ghost/.style={stage, draw=muted, dashed, fill=black!2, text=ink2,
                minimum height=0mm},
  flow/.style={-{Latex[length=2.6mm]}, line width=0.8pt, draw=ink2},
  ghostflow/.style={-{Latex[length=2.4mm]}, line width=0.7pt, draw=muted, dashed},
]

\def\colsep{51mm}   % centre-to-centre column pitch
\def\rowsep{-52mm}  % row 2 offset

% ------------------------------------------------------------- row 1 --------
\node[preblock] (prereg) at (0,0) {%
  {\head Frozen decision rules}\\[1pt]
  {\body STOP-gate $+$ paired-$\Delta$ win-rule frozen under a git tag
   before any result; each re-test's rule journaled before its run.}};

\node[runblock] (campaign) at (\colsep,0) {%
  {\head Controlled campaign}\\[1pt]
  {\body full CAFRL (11.97\,M) vs.\ the backbone-matched
   ``SpatialBase'' (4.0\,M) $+$ Xception / F3-Net / FreqNet, identical
   multi-CRF data.\\
   \textit{Confounds ruled out:} augmentation; capacity.}};

\node[stopblock] (stop) at (2*\colsep,0) {%
  {\head STOP-gate fires --- the initial configuration fails}\\[1pt]
  {\body SpatialBase scores above full CAFRL at every level
   (c40: .9590 vs.\ .9224; $\Delta$ +0.0366, CI [+0.020,\,+0.054]).}};

\node[ghost] (ablations) at (3*\colsep,0) {%
  {\head\color{ink2} Planned ablation campaign}\\[1pt]
  {\body six arm families, 38 runs --- e.g., GRL-vs-HSIC bake-off, conditioning,
   fusion, curriculum --- \textbf{never run} per the frozen rule.\\[1pt]
   \color{freq!70!black}\textbf{$\approx$270\,h of training saved}}};

% ------------------------------------------------------------- row 2 --------
\node[audblock] (audit) at (0,\rowsep) {%
  {\head Adversarial self-audit of the negative}\\[1pt]
  {\body quoted AUCs recompute exactly from stored scores; causal reading unsound.
   Four defects biased the comparison \emph{against} frequency: D1 auxiliary-loss
   dominance $\cdot$ D2 $\gamma$ deadlock $\cdot$ D3 optimization
   mismatch $\cdot$ D4 narrow trunk.}};

\node[runblock] (retests) at (\colsep,\rowsep) {%
  {\head Pre-specified fair re-tests (3 of 4; \mbox{repair D1--D4})}\\[1pt]
  {\body TrunkCtrl: trunk gap recipe-dominant\\
   Fair-N (no invariance losses):
   0.9586 vs.\ SpatialBase 0.9590\\
   Fair-W (full trunk $+\gamma$-residual): +0.0055, CI incl.\ 0\\[1pt]
   {\color{muted}$\approx$35\,h}}};

\node[runblock] (seeds) at (2*\colsep,\rowsep) {%
  {\head Seed confirmation (3 seeds vs.\ 3 seeds)}\\[1pt]
  {\body pooled $\Delta$ at c40: Fair-N $-$0.0009 / Fair-W +0.0022,
   both CIs include 0 --- no detectable difference.\\[1pt]
   {\color{muted}$\approx$31\,h}}};

\node[verblock] (verdict) at (3*\colsep,\rowsep) {%
  {\head Verdict}\\[1pt]
  {\body SpatialBase's training recipe recovered 3.96\,pt for CAFRL's own
   components (c40, s42); the
   block-DCT and FFT-phase streams: no detectable difference; training on real
   H.264 levels beat PSNR-matched JPEG by 7.48\,pt at c40 (averaged over
   c40 exposure; T4; raw weights), and including the c40-level data in training raised
   the mean single-model c40 AUC by 1.88\,pt averaged over codec (T5; raw
   weights); both registered, paired, \emph{under single-pass H.264 CRF
   re-encoding, in corpus}.}};

% -------------------------------------------------------------- edges -------

% wrap-around: STOP-gate down and back to the audit on row 2. The horizontal
% leg runs in the clear lane midway between the bottom edge of the tallest

\draw[flow] (seeds.east) -- (verdict.west);
